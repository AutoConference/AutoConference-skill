%% Palette: `tol-bright` (design-axes.md axis 14).
%% Paul Tol's 'bright' qualitative scheme (SRON technical note on colour schemes); colour-blind safe.
%% Redefines the family's semantic color tokens. Every block reads the tokens,
%% never a literal color, so a paper changes its whole visual key here and
%% nowhere else. Color is always a redundant cue: the tables still mark best
%% with \best and direction with \up/\down, so a greyscale print loses
%% nothing that carries meaning.

\definecolor{ACCyan}{HTML}{4477AA}   % secondary accent
\definecolor{ACLink}{HTML}{AA3377}   % project link only
\definecolor{ACInk}{HTML}{1F2A33}   % body-dark
\definecolor{ACMuted}{HTML}{6E7B85}   % rules, secondary text
\definecolor{ACGreen}{HTML}{228833}   % a delta in the good direction
\definecolor{ACWorse}{HTML}{EE6677}   % a delta in the bad direction
\definecolor{ACPrimary}{HTML}{EEDDF2}   % row tint: this study's family
\definecolor{ACCompare}{HTML}{FBDDE1}   % row tint: comparison family
\definecolor{ACNeutral}{HTML}{E7E7E7}   % row tint: reference family
\definecolor{ACRankOne}{HTML}{F5EFC8}   % cell background: best

\definecolor{ACRankThree}{HTML}{EEDDF2}   % cell background: third
