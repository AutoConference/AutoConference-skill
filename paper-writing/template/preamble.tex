% ------------------------------------------------------------------- icons
% The system figure in all seven source papers is not boxes and arrows. It is
% boxes and arrows WITH ICONS, and that is most of the difference between a
% figure a reader calls designed and one they call a flowchart. P1 puts a robot
% on the policy model, a snowflake on the frozen reference and a flame on the
% trained one. P7 draws four snowflakes, a flame beside the adapter it tunes,
% and keys both in a legend in the corner.
%
% FontAwesome 5 Free supplies them: vector glyphs that recolour with the
% palette, embedded in the PDF, with no external asset and no raster anywhere.
% The load is guarded, so a TeX tree without the package still builds a
% complete paper -- every icon simply renders as nothing. A missing glyph is
% worth less than a failed build.
\newif\ifACIcons\ACIconsfalse
\IfFileExists{fontawesome5.sty}{\usepackage{fontawesome5}\ACIconstrue}{}

% \acicon{Database} -- the glyph a node leads with, at the figure's icon size.
% An unknown name collapses to nothing rather than erroring, for the same
% reason. NOTE: \ifcsname is necessary but NOT sufficient -- fontawesome5
% defines deprecated aliases (\faExchange, \faShield) whose glyphs are not in
% the Free set and which error at use. Only names in references/figure-icons.md
% have been verified to render; check_paper_structure.py enforces that list.
\newcommand{\acicon}[1]{%
  \ifACIcons\ifcsname fa#1\endcsname{\large\color{ACIcon}\csname fa#1\endcsname}\fi\fi}
% the same glyph inline, at the surrounding text size, in a colour of your
% choosing: for legends, badges and anything sitting in a line of prose.
\newcommand{\aciconat}[2]{%
  \ifACIcons\ifcsname fa#2\endcsname{\color{#1}\csname fa#2\endcsname}\fi\fi}
\newcommand{\acsnow}{\aciconat{ACFrozen}{Snowflake}}
\newcommand{\acflame}{\aciconat{ACTrained}{Fire}}

% -------------------------------------------- glyphs FontAwesome does not have
% Two things the source figures draw that no icon font supplies. (A repeated
% stage is the `acdeck` style above, not a macro: it has to be a node so an
% arrow can land on it.) Each is a
% Tier C drawing: structure only, never a magnitude, never an image that does
% not exist (references/figure-ladder.md).
% \actokens{6}  a representation, drawn as a strip of tokens instead of named
% \acgrid{4}    an attention or score matrix, drawn as a small checkered grid
\newcommand{\actokens}[1]{%
  \begin{tikzpicture}[baseline=-0.55ex]
    \foreach \acti in {1,...,#1}{\node[actok] at (0.135*\acti,0) {};}
  \end{tikzpicture}}
\newcommand{\acgrid}[1]{%
  \begin{tikzpicture}[baseline=-0.55ex]
    \foreach \acgr in {1,...,#1}{\foreach \acgc in {1,...,#1}{%
      \pgfmathparse{mod(\acgr+\acgc,2) == 0 ? 30 : 8}%
      \node[draw=ACIcon!45,fill=ACIcon!\pgfmathresult,line width=0.25pt,
            minimum size=0.13cm,inner sep=0pt] at (0.13*\acgc,-0.13*\acgr) {};}}
  \end{tikzpicture}}
% A circled operator, the way P7 keys its two concatenations: (W) and (C).
\newcommand{\acop}[1]{%
  \tikz[baseline=-0.55ex]{\node[draw=ACIcon!75,fill=white,circle,line width=0.5pt,
    inner sep=0.6pt,font=\tiny\bfseries,text=ACIcon] {#1};}}

% -------------------------------------------------------- slot enforcement
% \slot{...} marks a value that evidence must supply.
%   draft build      -> grey bracketed marker, compiles
%   --final build    -> hard LaTeX error, build fails
% This is the mechanical guarantee that no placeholder reaches a submission.
\newif\ifACfinal\ACfinalfalse
\newcommand{\slot}[1]{%
  \ifACfinal
    \PackageError{acpaper}{Unfilled slot: #1}%
      {Every \string\slot\space must be replaced with verified evidence before a
       final build. See references/paper-writing-template.md section 9.}%
  \else
    \textcolor{ACMuted}{\textit{[#1]}}%
  \fi
}

% Whether a project mark sits left of the title is drawn (design-axes.md axis
% 16): P1, P3 and P7 carry one, P2 and P5 carry none and start the title at the
% margin. Every paper in this family used to carry the same mark.
% Scale a diagram to the float width, NEVER above its natural size. A plain
% \resizebox{\acfloatwidth}{!} enlarges a small diagram to full width, and a
% three-box figure stretched across a two-column page prints its labels larger
% than the body text -- which is what "the figure is suddenly huge and out of
% proportion" looks like from the reader's side.
\newcommand{\acfit}[1]{\resizebox{\ifdim\width>\acfloatwidth\acfloatwidth\else\width\fi}{!}{#1}}

% neurips_2025.sty renders \@author inside a plain `tabular[t]{c}` cell, whose
% `c` column has no width constraint: it sizes to the content's natural width
% and only centers it, so a normal-length affiliation line (institution name,
% \qquad, second institution) overflows the text block symmetrically on both
% sides. This is independent of the title, and independent of author-name
% length. Fixing it means bounding the content's own width before the style
% file ever sees it: wrap \author{...} in this, and the outer `c` column
% takes on the minipage's fixed, safe width and wraps long lines instead.
\newcommand{\AuthorBlock}[1]{%
  \begin{minipage}{0.92\textwidth}
    \centering
    #1
  \end{minipage}%
}

% ------------------------------------------------------------ tikz styles
% Tier C structural diagrams (references/figure-ladder.md).
\tikzset{
  acbox/.style={rounded corners=2.5pt,draw=ACBlue!55,line width=0.7pt,fill=white,inner sep=4pt},
  acsoft/.style={rounded corners=2.5pt,draw=ACCyan!65,line width=0.7pt,fill=ACPrimary!35,inner sep=4pt},
  acarrow/.style={-{Latex[length=2.2mm]},line width=0.8pt,draw=ACBlue!80},
  acnote/.style={font=\scriptsize,text=ACMuted,align=center},
  % --- vocabulary the overview-figure grammars share (design-axes.md axis 15).
  % A system figure in this family is Tier C: boxes, arrows, badges and labels.
  % It never implies a magnitude and never stands in for an image that does not
  % exist, so the same diagram is honest whether or not artwork was ever made.
  acstage/.style={acbox,align=center,font=\small,inner sep=5pt},
  acbadge/.style={rounded corners=1.5pt,inner sep=1.6pt,font=\tiny\bfseries,text=white},
  acfrozen/.style={acbadge,fill=ACMuted},
  actrained/.style={acbadge,fill=ACBlue},
  acgroupbox/.style={draw=ACMuted!55,dashed,rounded corners=3pt,inner sep=6pt},
  acpanel/.style={acbox,draw=ACMuted!45,align=left,font=\footnotesize,inner sep=5pt},
  acflow/.style={acsoft,align=left,font=\footnotesize,inner sep=4pt},
  % --- the richer vocabulary (design-axes.md axis 15) ---------------------
  % Measured off P1, P5 and P7. What separates their system figures from a
  % flowchart is not more boxes; it is an icon in each box, a legend that keys
  % the badges, a tint saying which region the paper contributed, and more
  % than one kind of shape and arrow. Each of those is a style here, so a
  % grammar spends its lines on the argument rather than on drawing.
  %
  % accard  a stage: an icon, a bold name, a muted line of what it does
  % acdeck  the same card as a deck, for a stage that repeats K times
  % actrap  a trapezium, the conventional shape for an encoder or decoder
  % acours  a tinted region: what this paper contributes
  % acbase  a grey region: what every compared arm already had
  % aclegend the key, bottom-left or bottom-right, never floating free
  % actag   the small uppercase label that names a region or a state
  % actok   one token of a representation, for \actokens
  % acstat  the second arrow language: statistics, not representations
  accard/.style={acbox,align=center,font=\scriptsize,inner sep=4pt,
                 minimum height=1.15cm,text width=2.0cm},
  acdeck/.style={accard,fill=white,
                 blur shadow={shadow blur steps=4,shadow xshift=1pt,
                              shadow yshift=-1pt,shadow opacity=22}},
  actrap/.style={trapezium,trapezium angle=68,draw=ACCyan!65,fill=ACCyan!8,
                 line width=0.7pt,align=center,font=\scriptsize,inner sep=3pt,
                 minimum height=1.1cm},
  acours/.style={rounded corners=4pt,draw=ACBlue!45,fill=ACPrimary!75,line width=0.6pt},
  acbase/.style={rounded corners=4pt,draw=ACMuted!40,fill=ACNeutral!60,line width=0.6pt},
  aclegend/.style={rounded corners=2.5pt,draw=ACMuted!50,fill=white,inner sep=5pt,
                   font=\scriptsize,align=left},
  actag/.style={font=\tiny\bfseries,text=ACMuted,inner sep=1pt},
  actok/.style={draw=ACIcon!70,fill=ACIcon!18,line width=0.35pt,
                minimum size=2.4pt,inner sep=0pt},
  acstat/.style={-{Latex[length=1.8mm]},line width=0.7pt,draw=ACCyan!85,
                 dash pattern=on 2.2pt off 1.6pt},
  % --- worked-instance vocabulary (design-axes.md axis 15, density pass) ---
  % What kept a denser figure from turning into noise, measured on the first
  % paper rebuilt this way (paper-batch-v2/S03-speculative-decoding): ONE
  % shared ruler of positions or items reused by every compared row, and a
  % cell's STATUS drawn as the cell's own fill and glyph rather than as a
  % neutral cell plus a badge beside it. These styles are that vocabulary, so
  % an overview grammar spends its lines on which items and which rows, never
  % on redeclaring what a cell looks like.
  %
  % acitem       a per-item cell at a ruler position; the four variants below
  %              ARE its status -- never leave a cell in the bare `acitem`
  %              style next to one that has a state, the way an unfilled
  %              table cell would read
  % acitemkeep   kept / correct / retained at this position
  % acitemdrop   dropped / rejected / incorrect at this position
  % acitemgate   the position where THIS PAPER's mechanism makes a call the
  %              other row does not (tint matches acours)
  % acitemna     not reached / excluded / not applicable at this position
  % acruler      the shared position or item label above or below the rows
  % acrowlabel   a row's own name, right-aligned against the ruler
  % acfeat       a small labelled node for one summand, statistic or
  %              sub-step: what a decision or a merge reads, drawn small
  %              enough that four or five sit in the space of one accard
  acitem/.style={draw=ACMuted!45,rounded corners=1pt,line width=0.5pt,
                 minimum width=0.62cm,minimum height=0.56cm,inner sep=0pt,
                 fill=white,align=center,font=\small},
  acitemkeep/.style={acitem,draw=ACGreen!60,fill=ACGreen!10},
  acitemdrop/.style={acitem,draw=BrickRed!60,fill=BrickRed!8},
  acitemgate/.style={acitem,draw=ACBlue!65,fill=ACPrimary!65},
  acitemna/.style={acitem,draw=ACMuted!35,fill=ACNeutral!30,densely dotted},
  acruler/.style={font=\tiny,text=ACMuted},
  acrowlabel/.style={font=\scriptsize,align=right,text width=1.9cm},
  acfeat/.style={draw=ACIcon!55,rounded corners=1.5pt,line width=0.5pt,
                 fill=ACPrimary!18,minimum width=1.4cm,minimum height=0.5cm,
                 align=center,font=\tiny,inner sep=1pt}
}
